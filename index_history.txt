   }

+function findOkButtonInAllFrames(

+  // 遍历所有 frames
+  try {
+    for (let i = 0; i < win.frames.length; i++) {
+      try {
+        const frame = win.frames[i];
+        if (frame && frame.document) {

+          if (btn) {

+          // 递归搜索嵌套的 iframes
+          const nestedBtn = findOkButtonInAllFrames(frame, selectors);
+          if (nestedBtn) {
+            return nestedBtn;
+          }
+        }
+      } catch (e) {
+        // 跨域 iframe 无法访问，静默忽略
+      }
+    }
+  } catch (e) {
+    // 无法访问 frames，静默忽略
+  }
+
+  return null;
